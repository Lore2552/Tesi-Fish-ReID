=== COMANDI PROGETTO TESI - Re-ID Melops ===


--- TRAINING ---

# Addestra sulla TESTA (head) per 5 epoche
py train.py --part head --epochs 5

# Addestra sul CORPO (body) per 5 epoche
py train.py --part body --epochs 5

# Opzioni aggiuntive (batch size personalizzato, piu' epoche)
py train.py --part head --epochs 10 --batch_size 16


--- VALUTAZIONE (Rank-1, mAP, CMC Curve) ---

# Valuta la run 1, usando il modello dell'epoca 5, sulla testa
py evaluate.py --run 1 --epoch 5 --part head

# Valuta la run 2, usando il modello dell'epoca 3, sul corpo
py evaluate.py --run 2 --epoch 3 --part body


--- HEATMAP COMPARATIVE (Explainability) ---

py compare_attention.py --run 1 --epoch 5 --part head
